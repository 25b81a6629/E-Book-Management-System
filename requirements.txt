        <input type="text" id="author" placeholder="Author Name">
        <button onclick="addBook()">Add Book</button>
        
        <ul id="bookList"></ul>
    </div>

    <!-- JavaScript Functionality -->
    <script>
        function addBook() {
            const titleInput = document.getElementById('title');
            const authorInput = document.getElementById('author');
            const bookList = document.getElementById('bookList');

            // Validate inputs
            if (titleInput.value.trim() === '' || authorInput.value.trim() === '') {
                alert('Please fill out both fields');
                return;
            }

            // Create list item element
            const li = document.createElement('li');
            li.innerHTML = `
                <span><strong>${titleInput.value}</strong> by ${authorInput.value}</span>
                <button class="delete-btn" onclick="deleteBook(this)">Delete</button>
            `;

            // Append to the list and clear inputs
            bookList.appendChild(li);
            titleInput.value = '';
            authorInput.value = '';
        }

        function deleteBook(button) {
            // Remove the parent li element of the clicked button
            button.parentElement.remove();
        }
    </script>
